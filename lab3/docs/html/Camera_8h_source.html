11// Helper class for working with elements of SE(3)

22 // Default constructor (R = I, r = 0)
23 Pose();
24
25 // Constructor from OpenCV rotation vector and translation vector
26 Pose(const cv::Mat & rvec, const cv::Mat & tvec);
27
28 // Group operation of SE(3)
29 //
30 // Tac = Tab*Tbc
31 //
32 // [ Rac, rCAa ] = [ Rab, rBAa ] * [ Rbc, rCBb ] = [ Rab*Rbc, Rab*rCBb + rBAa ]
33 // [ 0, 1 ] [ 0, 1 ] [ 0, 1 ] [ 0, 1 ]
34 //
35 Pose operator*(const Pose & other) const;
36
37 // Inverse element in SE(3)
38 //
39 // Tab^-1 = Tba
40 //
41 // [ Rab, rBAa ]^-1 = [ Rba, rABb ] = [ Rab^T, -Rab^T*rBAa ]
42 // [ 0, 1 ] [ 0, 1 ] [ 0, 1 ]
43 //
44 Pose inverse() const;
45};
