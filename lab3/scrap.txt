void Camera::calibrate(ChessboardData & chessboardData)
{
    std::vector<cv::Point3f> rPNn_all = chessboardData.chessboard.gridPoints();

    std::vector<std::vector<cv::Point2f>> rQOi_all;
    for (const auto & chessboardImage : chessboardData.chessboardImages)
    {
        rQOi_all.push_back(chessboardImage.corners);
    }
    //std::cout << "rQOi: (" << rQOi_all[0] << ", " << rQOi_all[1] << ")" << std::endl;

    assert(!rQOi_all.empty());

    imageSize = chessboardData.chessboardImages[0].image.size();
    
    flags = cv::CALIB_RATIONAL_MODEL | cv::CALIB_THIN_PRISM_MODEL;

    // Find intrinsic and extrinsic camera parameters
    cameraMatrix = cv::Mat::eye(3, 3, CV_64F);
    distCoeffs = cv::Mat::zeros(12, 1, CV_64F);
    std::vector<cv::Mat> Thetacn_all, rNCc_all;
    std::cout << "Calibrating camera..." << std::flush;
    std::cout << "rQOi: (" << rQOi_all[0] << ", " << rQOi_all[1] << ")" << std::endl;


    // TODO: Calibrate camera from detected chessboard corners
    double rms = cv::calibrateCamera(rPNn_all, rQOi_all, imageSize, cameraMatrix, distCoeffs, Thetacn_all, rNCc_all, flags);	

    std::cout << " done" << std::endl;
    
    // Calculate horizontal, vertical and diagonal field of view
    calcFieldOfView();

    // Write extrinsic camera parameters for each chessboard image
    assert(chessboardData.chessboardImages.size() == rNCc_all.size());
    assert(chessboardData.chessboardImages.size() == Thetacn_all.size());
    for (std::size_t k = 0; k < chessboardData.chessboardImages.size(); ++k)
    {
        // Set the camera orientation and position (extrinsic camera parameters)
        // Set the camera orientation and position (extrinsic camera parameters)
        Pose & Tnc = chessboardData.chessboardImages[k].Tnc;
        cv::Mat R;
        cv::Rodrigues(Thetacn_all[k], R);
        Tnc.rotationMatrix = cv::Mat(R.t());                     // Transpose because we want Rnc, not Rcn
        Tnc.translationVector = cv::Mat(-R.t() * rNCc_all[k]);   // Convert rCNc to rCNn
    }
    
    printCalibration();
    std::cout << std::setw(30) << "RMS reprojection error: " << rms << std::endl;

    assert(cv::checkRange(cameraMatrix));
    assert(cv::checkRange(distCoeffs));
}

      void ChessboardImage::drawBox(const Chessboard & chessboard, const Camera & camera)
{
    if (corners.empty() || !isFound) {
        std::cerr << "No chessboard corners detected. Cannot draw box." << std::endl;
        return;
    }

    // Find the bottom-left and top-right corners of the chessboard
    cv::Point2f bottomLeft = corners[0];
    cv::Point2f topRight = corners[corners.size() - 1];

    // Define the box dimensions
    double width = (chessboard.boardSize.width) * chessboard.squareSize;
    double height = (chessboard.boardSize.height) * chessboard.squareSize;
    double depth = 0.23; // 23 cm, height of the CodeNames box

    // Define the vertices of the box in world coordinates
    std::vector<cv::Point3d> boxVertices = {
        {0, 0, 0}, {width, 0, 0}, {width, height, 0}, {0, height, 0},
        {0, 0, depth}, {width, 0, depth}, {width, height, depth}, {0, height, depth}
    };

    // Define the edges of the box
    std::vector<std::pair<int, int>> edges = {
        {0, 1}, {1, 2}, {2, 3}, {3, 0}, // Bottom face
        {4, 5}, {5, 6}, {6, 7}, {7, 4}, // Top face
        {0, 4}, {1, 5}, {2, 6}, {3, 7}  // Vertical edges
    };

    // Project box vertices to image coordinates
    std::vector<cv::Point2d> projectedPoints;
    cv::projectPoints(boxVertices, Tnc.rotationMatrix, Tnc.translationVector, 
                      camera.cameraMatrix, camera.distCoeffs, projectedPoints);

    // Draw each edge of the box
    for (const auto& edge : edges)
    {
        cv::Point2d start = projectedPoints[edge.first];
        cv::Point2d end = projectedPoints[edge.second];

        // Check if both endpoints are within the image bounds
        if (start.x >= 0 && start.x < image.cols && start.y >= 0 && start.y < image.rows &&
            end.x >= 0 && end.x < image.cols && end.y >= 0 && end.y < image.rows)
        {
            // Draw the line
            cv::line(image, start, end, cv::Scalar(0, 255, 0), 2);
        }
    }
}



===================================== Working base Corner plotter =======================================
void ChessboardImage::drawBox(const Chessboard & chessboard, const Camera & camera)
{
    if (corners.empty() || !isFound) {
        std::cerr << "No chessboard corners detected. Cannot draw corners." << std::endl;
        return;
    }

    // Define the outer corners
    std::vector<cv::Point2f> outerCorners = {
        corners[0],                                                          // Top-left
        corners[chessboard.boardSize.width - 1],                             // Top-right
        corners[(chessboard.boardSize.height - 1) * chessboard.boardSize.width], // Bottom-left
        corners[corners.size() - 1]                                          // Bottom-right
    };

    // Color for the circles
    cv::Scalar color(0, 255, 0);  // Green

    // Draw circles on the outer corners
    for (const auto& corner : outerCorners)
    {
        // Convert corner to cv::Point as cv::circle expects this type
        cv::Point cornerPoint(static_cast<int>(std::round(corner.x)), 
                              static_cast<int>(std::round(corner.y)));
        
        // Draw the circle
        cv::circle(image, cornerPoint, 5, color, 2);  // 5 is the radius, 2 is the thickness
    }
}
