224 static Gaussian fromSqrtInfo(const Eigen::VectorX<Scalar> & nu, const Eigen::MatrixX<Scalar> & Xi)
225 {
226 assert(nu.size() == Xi.cols());
227 assert(Xi.rows() == Xi.cols());
228 Gaussian out;
229 // Solve Xi*mu = nu
230 out.mu_ = Xi.template triangularView<Eigen::Upper>().solve(nu);
231 // S = qr(Xi^{-T})
232 out.S_ = Xi.template triangularView<Eigen::Upper>().transpose().solve(
233 Eigen::MatrixX<Scalar>::Identity(Xi.rows(), Xi.cols())
234 );
235 Eigen::HouseholderQR<Eigen::Ref<Eigen::MatrixX<Scalar>>> qr(out.S_); // In-place QR decomposition
236 out.S_ = out.S_.template triangularView<Eigen::Upper>(); // Safe aliasing
237 return out;

455 static Gaussian fromSamples(const Eigen::MatrixX<Scalar> & X)
456 {
457 const Eigen::Index n = X.rows();
458 const Eigen::Index m = X.cols();
459
460 Gaussian out;
461
462 // Compute the sample mean
463 out.mu_ = X.rowwise().mean();
464
465 // Compute the sample square-root covariance
466 Eigen::MatrixX<Scalar> SS = std::sqrt(1.0 / (m - 1))*(X.colwise() - out.mu_).transpose();
467 Eigen::HouseholderQR<Eigen::Ref<Eigen::MatrixX<Scalar>>> qr(SS); // In-place QR decomposition
468 out.S_ = SS.topRows(n).template triangularView<Eigen::Upper>();
469 return out;

515template <typename IndexTypeA, typename IndexTypeB>
516Gaussian conditional(const IndexTypeA & idxA, const IndexTypeB & idxB, const Eigen::VectorX<Scalar> & xB) const
517{
518 Eigen::MatrixX<Scalar> SS(S_.rows(), idxB.size() + idxA.size());
519 SS << S_(Eigen::all, idxB), S_(Eigen::all, idxA);
520
521 Eigen::HouseholderQR<Eigen::MatrixX<Scalar>> qr(SS);
522 Eigen::MatrixX<Scalar> R = qr.matrixQR().template triangularView<Eigen::Upper>();
523
524 Eigen::Index nA = idxA.size();
525 Eigen::Index nB = idxB.size();
526
527 Eigen::MatrixX<Scalar> SBB = R.topLeftCorner(nB, nB);
528 Eigen::MatrixX<Scalar> SBA = R.topRightCorner(nB, nA);
529 Eigen::MatrixX<Scalar> SAA = R.bottomRightCorner(nA, nA);
530
531 Eigen::VectorX<Scalar> muA = mu_(idxA);
532 Eigen::VectorX<Scalar> muB = mu_(idxB);
533
534 Eigen::MatrixX<Scalar> Sc = SAA;
535 Eigen::VectorX<Scalar> muc = muA + SBA.transpose() * SBB.template triangularView<Eigen::Upper>().transpose().solve(xB - muB);
536
537 return Gaussian::fromSqrtMoment(muc, Sc);
538}

555 template <typename IndexTypeA, typename IndexTypeB>
556 Gaussian conditional(const IndexTypeA & idxA, const IndexTypeB & idxB, const Gaussian & pxB_y) const
557 {
558 const std::size_t & nA = idxA.size();
559 const std::size_t & nB = idxB.size();
560
561 // Form [S(:, idxB), S(:, idxA)]
562 Eigen::MatrixX<Scalar> SS(dim(), nA + nB);
563 SS << S_(Eigen::all, idxB), S_(Eigen::all, idxA);
564 // Q-less QR yields
565 // [S1, S2;
566 // 0, S3]
567 Eigen::HouseholderQR<Eigen::Ref<Eigen::MatrixX<Scalar>>> qr(SS); // In-place QR decomposition
568
569 // K = S1\S2
570 Eigen::MatrixX<Scalar> K = SS.topLeftCorner(nB, nB).template triangularView<Eigen::Upper>().solve(SS.topRightCorner(nB, nA));
571 Eigen::MatrixX<Scalar> S3 = SS.bottomRightCorner(nA, nA).template triangularView<Eigen::Upper>();
572
573 // Form [pxB_y.S*K; S3]
574 Eigen::MatrixX<Scalar> RR(nA + nB, nA);
575 RR << pxB_y.S_ * K,
576 S3;
577 // Q-less QR yields
578 // [R1;
579 // 0]
580 Eigen::HouseholderQR<Eigen::Ref<Eigen::MatrixX<Scalar>>> qr_RR(RR);
581
582 // p(x(idxA) | y) = N^-0.5(x(idxA); mu(idxA) + K.'*(pxB_y.mu - mu(idxB))), R1)
583 Gaussian out;
584 out.mu_ = mu_(idxA) + K.transpose()*(pxB_y.mean() - mu_(idxB));
585 out.S_ = RR.topRows(nA).template triangularView<Eigen::Upper>();
586 return out;
587 }

784 virtual bool isWithinConfidenceRegion(const Eigen::VectorX<Scalar> & x, double nSigma = 3.0) const override
785 {
786 // const Eigen::Index & n = dim();

788 // return false;

792 // Probability mass enclosed by nSigma standard deviations
793 double c = normcdf(nSigma) - normcdf(-nSigma);
794
795 // Squared radius in w coordinates

797
798 // Solve S.'*z = x - mu (triangular forward substitution)
799 Eigen::VectorX<Scalar> z = S_.transpose().template triangularView<Eigen::Lower>().solve(x - mu_);
800
801 // Check if the point is within the confidence region
