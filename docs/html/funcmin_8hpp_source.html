126template <typename Func>
127int NewtonTrustEig(Func costFunc, Eigen::VectorXd & x, Eigen::VectorXd & g, Eigen::MatrixXd & Q, Eigen::VectorXd & v, int verbosity = 0)
128{
129 typedef double Scalar;
130 typedef Eigen::VectorXd Vector;
131 typedef Eigen::MatrixXd Matrix;
132
133 assert(x.cols() == 1);
134 g.resize(x.size());
135 Q.resize(x.size(), x.size());
136 v.resize(x.size());
137
138 Matrix H(x.size(), x.size());
139
140 // Storage for trial point, gradient and Hessian
141 Vector xn(x.size());
142 Vector gn(x.size());
143 Matrix Hn(x.size(), x.size());
144
145 // Evaluate initial cost, gradient and Hessian
146 Scalar f = costFunc(x, g, H);
147 if (!std::isfinite(f) || !g.allFinite() || !H.allFinite()) // if any nan, -inf or +inf
148 {
149 if (verbosity > 1)
150 std::printf("ERROR: Initial point is not in domain of cost function\n");
151 return -1;
152 }
153
154 // Eigendecomposition of initial Hessian
155 Eigen::SelfAdjointEigenSolver<Matrix> eigenH(H);
156 v = eigenH.eigenvalues();
157 Q = eigenH.eigenvectors();
158
159 Scalar Delta = 1e0; // Initial trust-region radius
160
161 const int maxIterations = 5000;
162 for (int i = 0; i < maxIterations; ++i)
163 {
164 // Solve trust-region subproblem
165 Vector p;
166 trsEig(Q, v, g, Delta, p); // minimise 0.5*p.'*H*p + g.'*p subject to ||p|| <= Delta
167
168 Scalar pg = p.dot(g);
169 Scalar LambdaSq = -pg; // The Newton decrement squared is g.'*inv(H)*g = p.'*H*p
170 if (verbosity == 3)
171 std::printf("Iter = %5i, Cost = %10.2e, Newton decr^2 = %10.2e, Delta = %10.2e\n", i, f, LambdaSq, Delta);
172 if (verbosity == 1)
173 std::printf(".");
174
175 // const Scalar LambdaSqThreshold = std::sqrt(std::numeric_limits<Scalar>::epsilon()); // Loose convergence tolerance
176 const Scalar LambdaSqThreshold = 2*std::numeric_limits<Scalar>::epsilon(); // Tight convergence tolerance
177 if (std::fabs(LambdaSq) < LambdaSqThreshold)
178 {
179 if (verbosity >= 2)
180 std::printf("CONVERGED: Newton decrement below threshold in %i iterations\n", i);
181 return 0;
182 }
183
184 // Evaluate cost, gradient and Hessian for trial step
185 xn = x + p;
186 Scalar fn = costFunc(xn, gn, Hn);
187 bool outOfDomain = !std::isfinite(fn) || !gn.allFinite() || !Hn.allFinite(); // if any nan, -inf or +inf
188 Scalar rho;
189 if (outOfDomain)
190 {
191 rho = -1; // Force trust region reduction and reject step
192 }
193 else
194 {
195 Scalar dm = -pg - 0.5*p.dot(H*p); // Predicted reduction f - fp, where fp = f + p'*g + 0.5*p'*H*p
196 rho = (f - fn)/dm; // Actual reduction divided by predicted reduction
197 }
198
199 if (rho < 0.1)
200 {
201 Delta = 0.25*p.norm(); // Decrease trust region radius
202 }
203 else
204 {
205 if (rho > 0.75 && p.norm() > 0.8*Delta)
206 {
207 Delta = 2.0*Delta; // Increase trust region radius
208 }
209 }
210
211 if (rho >= 0.001)
212 {
213 // Accept the step
214 x = xn;
215 f = fn;
216 g = gn;
217 H = Hn;
218
219 // Eigendecomposition of accepted Hessian
220 Eigen::SelfAdjointEigenSolver<Matrix> eigenH_(H);
221 v = eigenH_.eigenvalues();
222 Q = eigenH_.eigenvectors();
223 }
224 }
225 if (verbosity > 1)
226 std::printf("WARNING: maximum number of iterations reached\n");
227 return 1;
228}

319template <typename Func>
320int SR1TrustEig(Func costFunc, Eigen::VectorXd & x, Eigen::VectorXd & g, Eigen::MatrixXd & Q, Eigen::VectorXd & v, int verbosity = 0)
321{
322 typedef double Scalar;
323 typedef Eigen::VectorXd Vector;
324 typedef Eigen::MatrixXd Matrix;
325
326 assert(x.cols() == 1);
327 g.resize(x.size());
328 Q.resize(x.size(), x.size());
329 v.resize(x.size());
330
331 Matrix H(x.size(),x.size());
332
333 // Storage for trial point and gradient
334 Vector xn(x.size());
335 Vector gn(x.size());
336
337 // Evaluate initial cost and gradient
338 Scalar f = costFunc(x, g);
339 if (!std::isfinite(f) || !g.allFinite()) // if any nan, -inf or +inf
340 {
341 if (verbosity > 1)
342 std::printf("ERROR: Initial point is not in domain of cost function\n");
343 return -1;
344 }
345
346 // Initial Hessian
347 H = Q*v.asDiagonal()*Q.transpose();
348
349 Scalar Delta = 1e0; // Initial trust-region radius
350
351 const int maxIterations = 5000;
352 for (int i = 0; i < maxIterations; ++i)
353 {
354 // Solve trust-region subproblem
355 Vector p;
356 trsEig(Q, v, g, Delta, p); // minimise 0.5*p.'*H*p + g.'*p subject to ||p|| <= Delta
357
358 Scalar pg = p.dot(g);
359 Scalar LambdaSq = -pg; // The Newton decrement squared is g.'*inv(H)*g = p.'*H*p
360 if (verbosity == 3)
361 std::printf("Iter = %5i, Cost = %10.2e, Newton decr^2 = %10.2e, Delta = %10.2e\n", i, f, LambdaSq, Delta);
362 if (verbosity == 1)
363 std::printf(".");
364
365 const Scalar LambdaSqThreshold = std::sqrt(std::numeric_limits<Scalar>::epsilon()); // Loose convergence tolerance
366 // const Scalar LambdaSqThreshold = 2*std::numeric_limits<Scalar>::epsilon(); // Tight convergence tolerance
367 if (std::fabs(LambdaSq) < LambdaSqThreshold && v(0) > 0.0)
368 {
369 if (verbosity >= 2)
370 std::printf("CONVERGED: Newton decrement below threshold in %i iterations\n", i);
371 return 0;
372 }
373
374 // Evaluate cost and gradient for trial step
375 xn = x + p;
376 Scalar fn = costFunc(xn, gn);
377 bool outOfDomain = !std::isfinite(fn) || !gn.allFinite(); // if any nan, -inf or +inf
378 Vector y = gn - g;
379
380 Scalar rho;
381 if (outOfDomain)
382 {
383 rho = -1; // Force trust region reduction and reject step
384 }
385 else
386 {
387 Scalar dm = -pg - 0.5*p.dot(H*p); // Predicted reduction f - fp, where fp = f + p'*g + 0.5*p'*H*p
388 rho = (f - fn)/dm; // Actual reduction divided by predicted reduction
389 }
390
391 if (rho < 0.1)
392 {
393 Delta = 0.25*p.norm(); // Decrease trust region radius
394 }
395 else
396 {
397 if (rho > 0.75 && p.norm() > 0.8*Delta)
398 {
399 Delta = 2.0*Delta; // Increase trust region radius
400 }
401 }
402
403 if (rho >= 0.001)
404 {
405 // Accept the step
406 x = xn;
407 f = fn;
408 g = gn;
409 }
410
411 // Update Hessian approximation
412 const Scalar sqrteps = std::sqrt(std::numeric_limits<Scalar>::epsilon());
413 if (!outOfDomain)
414 {
415 Vector w = y - H*p;
416 Scalar pw = p.dot(w);
417 if (std::fabs(pw) > sqrteps*p.norm()*w.norm())
418 {
419 Scalar s = sgn(pw);
420 Vector u = w.array()/std::sqrt(s*pw);
421 H = H + s*(u*u.transpose());
422
423 // Eigendecomposition of updated Hessian
424 Eigen::SelfAdjointEigenSolver<Matrix> eigenH(H);
425 v = eigenH.eigenvalues();
426 Q = eigenH.eigenvectors();
427 // We should do a rank-one update of Q and v instead of a full eigendecomposition using, e.g.,
428 // [1] Bunch, J.R., Nielsen, C.P. and Sorensen, D.C., 1978. Rank-one modification of the symmetric eigenproblem. Numerische Mathematik, 31(1), pp.31-48.
429 }
430 }
431 }
432 if (verbosity > 1)
433 std::printf("WARNING: maximum number of iterations reached\n");
434 return 1;
435}

519{

521 typedef Eigen::VectorXd Vector;
522 typedef Eigen::MatrixXd Matrix;
523
524 assert(x.cols() == 1);
525 g.resize(x.size());
526 assert(Xi.isUpperTriangular());
527
528 // Storage for trial point and gradient
529 Vector xn(x.size());
530 Vector gn(x.size());

532 // Evaluate initial cost and gradient

535 {
536 if (verbosity > 1)
537 std::printf("ERROR: Initial point is not in domain of cost function\n");
538 return -1;
539 }
540
541 Scalar Delta = 10e0; // Initial trust-region radius
542
543 const int maxIterations = 5000;
544 for (int i = 0; i < maxIterations; ++i)
545 {
546 // Solve trust-region subproblem
547 Vector p;
548 trsSqrt(Xi, g, Delta, p); // minimise 0.5*p.'*Xi.'*Xi*p + g.'*p subject to ||Xi*p|| <= Delta
549 Vector z = Xi*p;
550
551 Scalar pg = p.dot(g);
552 Scalar LambdaSq = -pg; // The Newton decrement squared is g.'*inv(H)*g = p.'*H*p = p.'*Xi.'*Xi*p
553 if (verbosity == 3)
554 std::printf("Iter = %5i, Cost = %10.2e, Newton decr^2 = %10.2e, Delta = %10.2e\n", i, f, LambdaSq, Delta);
555 if (verbosity == 1)
556 std::printf(".");
557
558 // const Scalar LambdaSqThreshold = std::sqrt(std::numeric_limits<Scalar>::epsilon()); // Loose convergence tolerance
559 const Scalar LambdaSqThreshold = 2*std::numeric_limits<Scalar>::epsilon(); // Tight convergence tolerance
560 if (std::fabs(LambdaSq) < LambdaSqThreshold)
561 {
562 if (verbosity >= 2)
563 std::printf("CONVERGED: Newton decrement below threshold in %i iterations\n", i);
564 return 0;
565 }
566
567 // Evaluate cost and gradient for trial step
568 xn = x + p;
569 Scalar fn = costFunc(xn, gn);
570 bool outOfDomain = !std::isfinite(fn) || !gn.allFinite(); // if any nan, -inf or +inf
571 Vector y = gn - g;
572
573 Scalar rho;
574 if (outOfDomain)
575 {
576 rho = -1; // Force trust region reduction and reject step
577 }
578 else
579 {
580 Scalar dm = -pg - 0.5*z.squaredNorm(); // Predicted reduction f - fp, where fp = f + p'*g + 0.5*p'*H*p
581 rho = (f - fn)/dm; // Actual reduction divided by predicted reduction
582 }
583
584 if (rho < 0.1)
585 {
586 Delta = 0.25*z.norm(); // Decrease trust region radius
587 }
588 else
589 {
590 if (rho > 0.75 && z.norm() > 0.8*Delta)
591 {
592 Delta = 2.0*Delta; // Increase trust region radius
593 }
594 }
595
596 if (rho >= 0.001)
597 {
598 // Accept the step
599 x = xn;
600 f = fn;
601 g = gn;
602 }
603
604 // Update Hessian approximation
605 const Scalar sqrteps = std::sqrt(std::numeric_limits<Scalar>::epsilon());
606 if (!outOfDomain)
607 {
608 Scalar py = p.dot(y);
609 if (py > sqrteps*y.norm()*p.norm())
610 {
611 // Form
612 // [ Xi*p, Xi ]
613 // [ 0, y^T/sqrt(y^T*p) ]
614 Matrix RR(Xi.rows() + 1, Xi.cols() + 1);
615 RR << Xi*p, Xi,
616 0, y.transpose()/std::sqrt(py);
617 // Q-less QR yields
618 // [ R1, R2 ]
619 // [ 0, R3 ]
620 Eigen::HouseholderQR<Eigen::Ref<Eigen::MatrixX<Scalar>>> qr(RR); // In-place QR decomposition
621 Xi = RR.bottomRightCorner(Xi.rows(), Xi.cols()).triangularView<Eigen::Upper>();
622 }
623 }
624 }
625 if (verbosity > 1)
626 std::printf("WARNING: maximum number of iterations reached\n");
627 return 1;
628}

768 Delta = 2.0*Delta; // Increase trust region radius
769 }

778 }

782 if (!outOfDomain)

784 Scalar py = p.dot(y);
785 if (py > sqrteps*y.norm()*p.norm())

799 for (typename Eigen::SparseMatrix<Scalar>::InnerIterator it(XiPiT, k); it; ++it)

801 // Copy y.transpose()/std::sqrt(py) into bottom row of XiPiT_yTsqrtpy
802 Eigen::RowVectorX<Scalar> yTsqrtpy = y.transpose()/std::sqrt(py);
803 for (int i = 0; i < Xi.cols(); ++i)
804 XiPiT_yTsqrtpy.insert(Xi.rows(), i) = yTsqrtpy(i);
805
806 // Z1.'*[Xi*Pi.'; y.'/realsqrt(py)]
807 Eigen::SparseMatrix<Scalar> Z1T_XiPiT_yTsqrtpy = (spqr_solver.matrixQ().transpose()*XiPiT_yTsqrtpy.toDense()).bottomRows(Xi.rows()).sparseView();
808
809 // Compute column-pivoting QR decomposition: Z1.'*[Xi*Pi.'; y.'/realsqrt(py)]*Pi2 = Q2*R2, where Pi2 reduces fill-in of R2
810 Z1T_XiPiT_yTsqrtpy.makeCompressed();
811 spqr_solver.compute(Z1T_XiPiT_yTsqrtpy);
812
813 Xi = spqr_solver.matrixR();
814 Pi = spqr_solver.colsPermutation();
815 }
816 }
817 }
818 if (verbosity > 1)
819 std::printf("WARNING: maximum number of iterations reached\n");
820 return 1;
821}

864template <typename Func>
865int BFGSTrustSqrtInv(Func costFunc, Eigen::VectorXd & x, Eigen::VectorXd & g, Eigen::MatrixXd & S, int verbosity = 0)
866{
867 typedef double Scalar;
868 typedef Eigen::VectorXd Vector;
869 typedef Eigen::MatrixXd Matrix;
870
871 assert(x.cols() == 1);
872 g.resize(x.size());
873 assert(S.isUpperTriangular());
874
875 // Storage for trial point and gradient
876 Vector xn(x.size());
877 Vector gn(x.size());
878
879 // Evaluate initial cost and gradient
880 Scalar f = costFunc(x, g);
881 if (!std::isfinite(f) || !g.allFinite()) // if any nan, -inf or +inf
882 {
883 if (verbosity > 1)
884 std::printf("ERROR: Initial point is not in domain of cost function\n");
885 return -1;
886 }
887
888 Scalar Delta = 10e0; // Initial trust-region radius
889
890 const int maxIterations = 5000;
891 for (int i = 0; i < maxIterations; ++i)
892 {
893 // Solve trust-region subproblem
894 Vector p;
895 trsSqrtInv(S, g, Delta, p); // minimise 0.5*p.'*inv(S.'*S)*p + g.'*p subject to ||inv(S.')*p|| <= Delta
896 // Solve S.'*z = p for z
897 Vector z = S.triangularView<Eigen::Upper>().transpose().solve(p);
898
899 Scalar pg = p.dot(g);
900 Scalar LambdaSq = -pg; // The Newton decrement squared is g.'*inv(H)*g = p.'*H*p = p.'*inv(S.'*S)*p
901 if (verbosity == 3)
902 std::printf("Iter = %5i, Cost = %10.2e, Newton decr^2 = %10.2e, Delta = %10.2e\n", i, f, LambdaSq, Delta);
903 if (verbosity == 1)
904 std::printf(".");
905
906 // const Scalar LambdaSqThreshold = std::sqrt(std::numeric_limits<Scalar>::epsilon()); // Loose convergence tolerance
907 const Scalar LambdaSqThreshold = 2*std::numeric_limits<Scalar>::epsilon(); // Tight convergence tolerance
908 if (std::fabs(LambdaSq) < LambdaSqThreshold)
909 {
910 if (verbosity >= 2)
911 std::printf("CONVERGED: Newton decrement below threshold in %i iterations\n", i);
912 return 0;
913 }
914
915 // Evaluate cost and gradient for trial step
916 xn = x + p;
917 Scalar fn = costFunc(xn, gn);
918 bool outOfDomain = !std::isfinite(fn) || !gn.allFinite(); // if any nan, -inf or +inf
919 Vector y = gn - g;
920
921 Scalar rho;
922 if (outOfDomain)
923 {
924 rho = -1; // Force trust region reduction and reject step
925 }
926 else
927 {
928 Scalar dm = -pg - 0.5*z.squaredNorm(); // Predicted reduction f - fp, where fp = f + p'*g + 0.5*p'*H*p
929 rho = (f - fn)/dm; // Actual reduction divided by predicted reduction
930 }
931
932 if (rho < 0.1)
933 {
934 Delta = 0.25*z.norm(); // Decrease trust region radius
935 }
936 else
937 {
938 if (rho > 0.75 && z.norm() > 0.8*Delta)
939 {
940 Delta = 2.0*Delta; // Increase trust region radius
941 }
942 }
943
944 if (rho >= 0.001)
945 {
946 // Accept the step
947 x = xn;
948 f = fn;
949 g = gn;
950 }
951
952 // Update Hessian approximation
953 const Scalar sqrteps = std::sqrt(std::numeric_limits<Scalar>::epsilon());
954 if (!outOfDomain)
955 {
956 Scalar py = p.dot(y);
957 if (py > sqrteps*y.norm()*p.norm())
958 {
959 // Form
960 // [ p^T/sqrt(y^T*p) ]
961 // [ S*(I - y*p^T/(y^T*p)) ]
962 Matrix RR(x.size() + 1, x.size());
963 RR << p.transpose()/std::sqrt(py),
964 S*(Eigen::MatrixXd::Identity(x.size(), x.size()) - y*p.transpose()/py);
965 // Q-less QR
966 Eigen::HouseholderQR<Eigen::Ref<Eigen::MatrixX<Scalar>>> qr(RR); // In-place QR decomposition
967 S = RR.topRows(x.size()).triangularView<Eigen::Upper>();
968 }
969 }
970 }
971 if (verbosity > 1)
972 std::printf("WARNING: maximum number of iterations reached\n");
973 return 1;
974}
