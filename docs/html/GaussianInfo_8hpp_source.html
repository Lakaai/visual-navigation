381 static GaussianInfo fromSamples(const Eigen::MatrixX<Scalar> & X)
382 {
383 const Eigen::Index n = X.rows();
384 const Eigen::Index m = X.cols();
385
386 // Compute the sample mean
387 Eigen::VectorXd mu = X.rowwise().mean();
388
389 // Compute the sample square-root covariance
390 Eigen::MatrixX<Scalar> SS = std::sqrt(1.0 / (m - 1))*(X.colwise() - mu).transpose();
391 Eigen::HouseholderQR<Eigen::Ref<Eigen::MatrixX<Scalar>>> qr(SS); // In-place QR decomposition
392 Eigen::MatrixXd S = SS.topRows(n).template triangularView<Eigen::Upper>();
393 return GaussianInfo::fromSqrtMoment(mu, S);

463 {
464 const std::size_t & n = dim();
465 std::vector<bool> isNotInIdx(n, true);
466 for (Eigen::Index ii = 0; ii < idx.size(); ++ii)

469 isNotInIdx[i] = false;

479 idxNot.push_back(i);

603 template <typename Func>
604 GaussianInfo affineTransform(Func h) const
605 {
606 Eigen::MatrixX<Scalar> J;
607 Eigen::VectorX<Scalar> mux = mean();
608 Eigen::VectorX<Scalar> muy = h(mux, J); // Evaluate function at mean value
609 const std::size_t m = J.rows();
610 const std::size_t n = J.cols();
611 assert(m == muy.size());
612 assert(n == dim());
613
614 // Linearise y = h(x) about x = mux
615 // y ~= h(mux) + J*(x - mux)
616 // = J*x + h(mux) - J*mux
617
618 // SVD approach for robustness
619 Eigen::JacobiSVD<Eigen::MatrixX<Scalar>> svd(J, Eigen::ComputeFullU | Eigen::ComputeFullV);
620 Eigen::VectorX<Scalar> s = svd.singularValues();
621 Scalar tol = std::max(m, n) * std::numeric_limits<Scalar>::epsilon() * s.maxCoeff();
622 std::size_t r = (s.array() > tol).count(); // Rank
623
624 Eigen::VectorX<Scalar> s1 = s.head(r);
625 Eigen::MatrixX<Scalar> U1 = svd.matrixU().leftCols(r);
626 Eigen::MatrixX<Scalar> U2 = svd.matrixU().rightCols(m - r);
627 Eigen::MatrixX<Scalar> V1 = svd.matrixV().leftCols(r);
628 Eigen::MatrixX<Scalar> V2 = svd.matrixV().rightCols(n - r);
629 Eigen::MatrixX<Scalar> Jp = V1 * (s1.asDiagonal().inverse()) * U1.transpose();
630
631 Eigen::MatrixX<Scalar> X = Xi_ * V2;
632 Eigen::MatrixX<Scalar> Y = Xi_ * Jp;
633 Eigen::VectorX<Scalar> b = muy - J * mux;
634 // std::cout << "Variable muy:" << std::endl;
635 // std::cout << muy << std::endl;
636 // Corrected sigma_max_ub calculation
637 Eigen::MatrixX<Scalar> XY(X.rows(), X.cols() + Y.cols());
638 XY << X, Y;
639 Scalar sigma_max_ub = std::sqrt(XY.array().square().sum());
640 Scalar kappa = 1e7 * sigma_max_ub;
641
642 // Build QR matrix given in equation 41
643 Eigen::MatrixX<Scalar> QR(n - r + m, n - r + m + 1);
644 QR.topRows(n) << X, Y, nu_ + Y * b;
645 QR.bottomRows(m - r) << Eigen::MatrixX<Scalar>::Zero(m - r, n - r),
646 kappa * U2.transpose(),
647 kappa * U2.transpose() * b;
648 // Q-less QR yields
649 // [R1, R2, nu1;
650 // 0, R3, nu2];
651 Eigen::MatrixX<Scalar> RR(n - r + m, n - r + m + 1);
652 Eigen::HouseholderQR<Eigen::Ref<Eigen::MatrixX<Scalar>>> qr(QR);
653 RR = qr.matrixQR();
654 // Print RR

657 // std::cout << "RR dimensions: " << RR.rows() << " x " << RR.cols() << std::endl;

659
660 // p(y) = N^-0.5(y; r + R*(h(mux) - J*mux), R)
661 GaussianInfo out(m);

663 // Extract R3 (m x m matrix, left of nu2)
664 Eigen::MatrixX<Scalar> R3 = RR.block(n - r, n - r, m, m).template triangularView<Eigen::Upper>();
665
666 // Extract nu2 (m x 1 matrix, last column of the bottom m rows)

814 virtual bool isWithinConfidenceRegion(const Eigen::VectorX<Scalar> & x, double nSigma = 3.0) const override
815 {
816 const Eigen::Index & n = dim();
817 assert(x.size() == n);
818
819 // Probability mass enclosed by nSigma standard deviations
820 double c = normcdf(nSigma) - normcdf(-nSigma);
821
822 // Squared radius in w coordinates
823 double r2 = chi2inv(c, n);
824
825 // Compute w = Xi * (x - mu)
826 // First, calculate mu by solving Xi * mu = nu
827 Eigen::VectorX<Scalar> mu = Xi_.template triangularView<Eigen::Upper>().solve(nu_);
828
829 // Now compute w
830 Eigen::VectorX<Scalar> w = Xi_ * (x - mu);
831
832 // Check if the point is within the confidence region
833 return w.squaredNorm() <= r2;
834 }

848 Eigen::Matrix4<Scalar> quadricSurface(double nSigma = 3.0) const

853 Scalar r2 = chi2inv(c, dim());

865 Q(3,3) = nu_.dot(nu_) - r2;
