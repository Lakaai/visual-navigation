54template <typename Scalar>
55Scalar MeasurementIdenticalTagBundle::logLikelihood(const Eigen::VectorX<Scalar> & x, const SystemEstimator & system) const
56{
57 const SystemSLAM & systemSLAM = dynamic_cast<const SystemSLAM &>(system);
58
59 Scalar logLikelihood = Scalar(0);
60 const Scalar tagLength = Scalar(0.166); // 166 mm edge length of each marker
61
62 // Extract body pose from the state vector
63 Eigen::Vector3<Scalar> rBNn = x.template segment<3>(6);
64 Eigen::Vector3<Scalar> ThetaNb = x.template segment<3>(9);
65 Eigen::Matrix3<Scalar> Rnb = rpy2rot(ThetaNb);
66 Pose<Scalar> Tnb(Rnb, rBNn);
67
68 // Calculate camera pose using body pose and body-to-camera transformation
69 Pose<Scalar> Tnc = Tnb * camera_.Tbc;
70
71 for (int i = 0; i < Y_.cols(); ++i) {
72 int markerId = markerIds_[i];
73 size_t landmarkIndex = systemSLAM.getLandmarkIndex(markerId);
74
75 std::size_t idx = systemSLAM.landmarkPositionIndex(landmarkIndex);
76 Eigen::Matrix<Scalar, 6, 1> landmark = x.template segment<6>(idx);
77 Eigen::Vector3<Scalar> rjNn = landmark.template head<3>();
78 Eigen::Vector3<Scalar> Thetanj = landmark.template tail<3>();
79 Eigen::Matrix3<Scalar> Rnj = rpy2rot(Thetanj);
80
81 for (int c = 0; c < 4; ++c) {
82 Eigen::Vector3<Scalar> rjcjj;
83 switch (c) {
84 case 0: rjcjj << -tagLength/2, tagLength/2, Scalar(0); break;
85 case 1: rjcjj << tagLength/2, tagLength/2, Scalar(0); break;
86 case 2: rjcjj << tagLength/2, -tagLength/2, Scalar(0); break;
87 case 3: rjcjj << -tagLength/2, -tagLength/2, Scalar(0); break;
88 }
89
90 Eigen::Vector3<Scalar> rjcNn = Rnj * rjcjj + rjNn;
91 Eigen::Vector2<Scalar> predictedCorner = camera_.worldToPixel(rjcNn, Tnb);
92 Eigen::Vector2<Scalar> measuredCorner(Scalar(Y_(2*c, i)), Scalar(Y_(2*c+1, i)));
93 Eigen::Vector2<Scalar> error = measuredCorner - predictedCorner;
94 logLikelihood += -Scalar(0.5) * error.squaredNorm() / (sigma_ * sigma_) - Scalar(std::log(2 * M_PI * sigma_ * sigma_));
95 }
96 }
97
98 // Add penalty for unassociated visible landmarks
99 size_t unassociatedLandmarks = visibleLandmarks_.size() - associatedLandmarks_.size();
100 Scalar imageArea = Scalar(camera_.imageSize.width * camera_.imageSize.height);
101 GaussianInfo<Scalar> tempGaussian = GaussianInfo<Scalar>::fromMoment(Eigen::Matrix<Scalar, 1, 1>(imageArea), Eigen::Matrix<Scalar, 1, 1>::Constant(1e-6));
102 Scalar logImageArea = tempGaussian.log(Eigen::Matrix<Scalar, 1, 1>(imageArea));
103 logLikelihood -= Scalar(4 * unassociatedLandmarks) * logImageArea;
104
105 return logLikelihood;
106}

109template <typename Scalar>
110Eigen::Vector2<Scalar> MeasurementIdenticalTagBundle::predictFeature(const Eigen::VectorX<Scalar> & x, const SystemSLAM & system, std::size_t idxLandmark) const
111{
112 // Obtain camera pose from state
113 Pose<Scalar> Tnc;
114 Tnc.translationVector = system.cameraPosition(camera_, x); // rCNn
115 Tnc.rotationMatrix = system.cameraOrientation(camera_, x); // Rnc
116
117 // Obtain landmark position from state
118 std::size_t idx = system.landmarkPositionIndex(idxLandmark);
119 Eigen::Vector3<Scalar> rPNn = x.template segment<3>(idx);
120
121 // Camera vector
122 Eigen::Vector3<Scalar> rPCc;
123 rPCc = Tnc.rotationMatrix.transpose() * (rPNn - Tnc.translationVector);
124
125 // Pixel coordinates
126 Eigen::Vector2<Scalar> rQOi;
127 rQOi = camera_.vectorToPixel(rPCc);
128
129 return rQOi;
130}

132template <typename Scalar>
133Eigen::VectorX<Scalar> MeasurementIdenticalTagBundle::predictFeatureBundle(const Eigen::VectorX<Scalar> & x, const SystemSLAM & system, const std::vector<std::size_t> & idxLandmarks) const
134{
135 // Ensure the state vector size matches the system's dimension
136 assert(x.size() == system.density.dim());
137
138 // Get the number of landmarks
139 const std::size_t & nL = idxLandmarks.size();
140
141 // Initialize the output vector h to store all predicted features
142 // The size is 2*nL because each feature has 2 coordinates (x and y in the image)
143 Eigen::VectorX<Scalar> h(2*nL);
144
145 // Iterate through each landmark
146 for (std::size_t i = 0; i < nL; ++i)
147 {
148 // Predict the feature location for the current landmark
149 Eigen::Vector2<Scalar> rQOi = predictFeature(x, system, idxLandmarks[i]);
150
151 // Set the pair of elements in h corresponding to this landmark
152 // We use 2*i as the starting index because each feature occupies 2 elements
153 h.template segment<2>(2*i) = rQOi;
154 }
155
156 // Return the complete feature bundle prediction vector
157 return h;
158}
